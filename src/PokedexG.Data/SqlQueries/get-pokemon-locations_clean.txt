select 
	encounters.pokemon_id as PokemonId,
	Min(encounters.min_level) as MinLevel,
	Max(encounters.max_level) as MaxLevel,
	Min(encounter_slots.rarity) as Rarity,
	region_names.name as RegionName,
	location_names.name as LocationName,
	encounter_methods.identifier as EncounterMethod,
	language_names.name as LanguageName,
	version_names.name as VersionName
	
from encounters
inner join versions
	on versions.id = encounters.version_id
inner join encounter_slots
	on encounter_slots.id = encounters.encounter_slot_id
inner join location_areas
	on location_areas.id = encounters.location_area_id
inner join locations
	on locations.id = location_areas.location_id
inner join region_names
	on region_names.region_id = locations.region_id
inner join location_names
	on location_names.location_id = location_areas.location_id
	and location_names.local_language_id = region_names.local_language_id
inner join encounter_methods
	on encounter_methods.id = encounter_slots.encounter_method_id
inner join language_names
	on language_names.language_id = region_names.local_language_id
	and language_names.local_language_id = region_names.local_language_id
inner join version_names
	on version_names.version_id = versions.id
	and version_names.local_language_id = region_names.local_language_id
	
where 
	encounters.version_id = 24 and 
	encounters.pokemon_id = @pokemonId and
	region_names.local_language_id = @languageId
	
group by 
	location_areas.id
