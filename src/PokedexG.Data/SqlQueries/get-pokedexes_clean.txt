-- RECONSTRUCTION 2026-07-23 : jointures réordonnées — le SQLite embarqué en 2016 tolérait
-- qu'un ON référence une table jointe plus loin (« ON clause references tables to its right »),
-- le SQLite moderne le refuse. Colonnes, conditions et tri inchangés ; résultat identique.
select
	pokedexes.id as Id,
	pokedexes.identifier as Identifier,
	region_names.name as Region,
	pokedex_prose.name as Name,
	pokedex_prose.description as Description,
	version_groups.identifier as VersionIdentifier,
	generation_names.generation_id as GenerationId,
	generation_names.name as GenerationName,
	language_names.name as LanguageName

from pokedexes
left join pokedex_prose
	on pokedex_prose.pokedex_id = pokedexes.id
left join pokedex_version_groups
	on pokedex_version_groups.pokedex_id = pokedexes.id
left join version_groups
	on version_groups.id = pokedex_version_groups.version_group_id
left join generation_names
	on generation_names.generation_id = version_groups.generation_id
	and generation_names.local_language_id = pokedex_prose.local_language_id
left join regions
	on regions.id = pokedexes.region_id
left join region_names
	on region_names.region_id = pokedexes.region_id
	and region_names.local_language_id = generation_names.local_language_id
left join language_names
	on language_names.language_id = pokedex_prose.local_language_id
	and language_names.local_language_id = pokedex_prose.local_language_id

where
	pokedexes.is_main_series = 1  and
	pokedex_prose.local_language_id = @languageId and
	(region_names.local_language_id = @languageId or region_names.local_language_id IS NULL)

order by
	version_groups.[order]
