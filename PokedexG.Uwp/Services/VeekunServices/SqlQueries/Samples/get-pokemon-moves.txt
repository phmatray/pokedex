select 
	pokemon_moves.pokemon_id as Id,
	--pokemon_moves.version_group_id,
	--pokemon_moves.move_id,
	--pokemon_moves.pokemon_move_method_id,
	pokemon_moves.[level] as [Level],
	--pokemon_moves.[order],
	--pokemon_move_methods.id,
	pokemon_move_methods.identifier as MoveMethod,
	--moves.id,
	--moves.identifier,
	--moves.generation_id,
	moves.type_id as TypeId,
	moves.power as Power,
	moves.pp as Pp,
	moves.accuracy as Accuracy,
	moves.priority as Priority,
	--moves.target_id,
	--moves.damage_class_id,
	--moves.effect_id,
	moves.effect_chance as EffectChance,
	--moves.contest_type_id,
	--moves.contest_effect_id,
	--moves.super_contest_effect_id,
	--move_names.move_id,
	--move_names.local_language_id,
	move_names.name as Name,
	(
		select move_names.name
		from move_names
		where move_names.local_language_id = 9 
		and move_names.move_id = moves.id
	) as NameEn,
	--move_flavor_text.move_id,
	--move_flavor_text.version_group_id,
	--move_flavor_text.language_id,
	move_flavor_text.flavor_text as FlavorText,
	--move_meta_ailment_names.move_meta_ailment_id,
	--move_meta_ailment_names.local_language_id,
	move_meta_ailment_names.name as Ailment,
	--move_meta_categories.id,
	move_meta_categories.identifier as Category,
	--move_meta.move_id,
	--move_meta.meta_category_id,
	move_meta.meta_ailment_id as AilmentId,
	move_meta.min_hits as MinHits,
	move_meta.max_hits as MaxHits,
	move_meta.min_turns as MinTurns,
	move_meta.max_turns as MaxTurns,
	move_meta.drain as Drain,
	move_meta.healing as Healing,
	move_meta.crit_rate as CritRate,
	move_meta.ailment_chance as AilmentChance,
	move_meta.flinch_chance as FlinchChance,
	move_meta.stat_chance as StatChance,
	types.id as TypeId,
	--types.identifier,
	--types.generation_id,
	--types.damage_class_id,
	--type_names.type_id,
	--type_names.local_language_id,
	type_names.name as Type,
	--generations.id,
	--generations.identifier,
	--generations.main_region_id,
	--generation_names.generation_id,
	--generation_names.local_language_id,
	generation_names.name as Generation,
	--regions.id,
	--regions.identifier,
	--region_names.region_id,
	--region_names.local_language_id,
	region_names.name as Region,
	--language_names.language_id,
	--language_names.local_language_id,
	language_names.name as Language,
	--move_damage_classes.id,
	move_damage_classes.identifier as DamageClass,
	--contest_type_names.contest_type_id,
	--contest_type_names.local_language_id,
	contest_type_names.name as ContestTypeName,
	contest_type_names.flavor as PokeblocFlavor,
	contest_type_names.color as PokeblocColor,
	--version_groups.id,
	version_groups.identifier as VersionGroup,
	--version_groups.generation_id,
	--version_groups.[order]
	--move_targets.id,
	move_targets.identifier as Targets
	
from pokemon_moves
inner join pokemon_move_methods
	on pokemon_move_methods.id = pokemon_moves.pokemon_move_method_id
inner join moves
	on moves.id = pokemon_moves.move_id
inner join move_names
	on move_names.move_id = pokemon_moves.move_id
	and move_names.local_language_id = language_names.language_id
inner join move_flavor_text
	on move_flavor_text.move_id = pokemon_moves.move_id
	and move_flavor_text.language_id = language_names.language_id
	and move_flavor_text.version_group_id = pokemon_moves.version_group_id
inner join move_meta_ailment_names
	on move_meta_ailment_names.move_meta_ailment_id = move_meta.meta_ailment_id
	and move_meta_ailment_names.local_language_id = language_names.language_id
inner join move_meta_categories
	on move_meta_categories.id = move_meta.meta_category_id
inner join move_meta
	on move_meta.move_id = pokemon_moves.move_id
inner join types
	on types.id = moves.type_id
inner join type_names
	on type_names.type_id = types.id
	and type_names.local_language_id = language_names.language_id
inner join generations
	on generations.id = moves.generation_id
inner join generation_names
	on generation_names.generation_id = moves.generation_id
	and generation_names.local_language_id = language_names.language_id
inner join regions
	on regions.id = generations.main_region_id
inner join region_names
	on region_names.region_id = regions.id
	and region_names.local_language_id = language_names.language_id
inner join language_names
	on language_names.language_id = move_names.local_language_id
	and language_names.local_language_id = move_names.local_language_id
inner join move_damage_classes
	on move_damage_classes.id = types.damage_class_id
inner join contest_type_names
	on contest_type_names.contest_type_id = moves.contest_type_id
	and contest_type_names.local_language_id = language_names.language_id
inner join version_groups
	on version_groups.id = pokemon_moves.version_group_id
inner join move_targets
	on move_targets.id = moves.target_id
	
where 
	pokemon_moves.pokemon_id = 1 and 
	pokemon_moves.pokemon_move_method_id = 1 and
	pokemon_moves.version_group_id = 16 and
	language_names.language_id = 5 
	
order by 
	pokemon_moves.level, pokemon_moves.[order]
