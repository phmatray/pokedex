select
	pokemon_egg_groups.species_id as PokemonId,
	--pokemon_egg_groups.egg_group_id,
	egg_groups.id as EggGroupId,
	egg_groups.identifier as EggGroupIdentifier,
	--egg_group_prose.egg_group_id,
	--egg_group_prose.local_language_id,
	egg_group_prose.name as EggGroupName
	
from pokemon_egg_groups
join egg_groups
	on egg_groups.id = pokemon_egg_groups.egg_group_id
join egg_group_prose
	on egg_group_prose.egg_group_id = pokemon_egg_groups.egg_group_id

where
	pokemon_egg_groups.species_id = 1 and
	egg_group_prose.local_language_id = 5
