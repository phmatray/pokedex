select
	types.id as Id,
	types.identifier as Identifier,
	--types.generation_id,
	--types.damage_class_id,
	--type_names.type_id,
	--type_names.local_language_id,
	type_names.name as Name
	
from types
join type_names
	on type_names.type_id = types.id

where
	types.id <= 10000 and
	type_names.local_language_id = 5
